
    2. **src/lib/fonts.ts** -- Create font configuration:

       ```typescript
       import { Playfair_Display, Inter } from 'next/font/google'

       export const playfairDisplay = Playfair_Display({
         subsets: ['latin'],
         display: 'swap',
         variable: '--font-playfair',
       })

       export const inter = Inter({
         subsets: ['latin'],
         display: 'swap',
         variable: '--font-inter',
       })
       ```

       NOTE: The CSS variable names `--font-playfair` and `--font-inter` are the next/font variables. The `@theme` block references these via `var(--font-playfair)` in the `--font-serif` and `--font-sans` definitions. This wiring is critical -- next/font sets the CSS variables on the html element, and Tailwind's `font-serif`/`font-sans` utilities reference them.

    3. **src/lib/utils.ts** -- Create utility functions:

       ```typescript
       import { type ClassValue, clsx } from 'clsx'
       import { twMerge } from 'tailwind-merge'

       export function cn(...inputs: ClassValue[]) {
         return twMerge(clsx(inputs))
       }
       ```

    4. **src/app/layout.tsx** -- Wire fonts into root layout:

       ```typescript
       import type { Metadata } from 'next'
       import { playfairDisplay, inter } from '@/lib/fonts'
       import './globals.css'

       export const metadata: Metadata = {
         title: 'Ikigai Consulting Group | Organizational Architects',
         description: 'Organizational Architects for Purpose-Driven Organizations in Ontario.',
       }

       export default function RootLayout({
         children,
       }: {
         children: React.ReactNode
       }) {
         return (
           <html lang="en" className={`${playfairDisplay.variable} ${inter.variable}`}>
             <body>{children}</body>
           </html>
         )
       }
       ```
